# PLMS Redesign — Phase 0 (Foundation) + Phase 1 (Masters) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bootstrap two new projects (`plms-api` and `plms-web`) outside the existing codebase, wire them together with Sanctum SPA auth, reverse-engineer the live PostgreSQL schema into fresh migrations and models, and deliver full CRUD API + UI for the Masters module (Buildings, Units, Employees, and all lookup tables).

**Architecture:** Laravel 11 REST API (`plms-api`) serves JSON over `/api/v1/` with Sanctum cookie auth and Spatie permission-based route guards. Next.js 15 (`plms-web`) consumes the API via Axios with React Query for server state and Zustand for client state. Both live at `C:\laragon\www\` alongside the untouched `plms_backup`.

**Tech Stack:** PHP 8.4, Laravel 11, Sanctum, Spatie Permission, nwidart/laravel-modules, kitloong/laravel-migrations-generator, Node v24, Next.js 15 App Router, TypeScript strict, Ant Design 5, Tailwind CSS v4, TanStack React Query v5, Zustand v5, Axios.

- [ ] **Step 3: Create the ApiResponse trait**

Create `app/Traits/ApiResponse.php`:
```php
<?php

namespace App\Traits;

use Illuminate\Http\JsonResponse;

trait ApiResponse
{
    protected function success(mixed $data = null, string $message = 'Success', int $status = 200): JsonResponse
    {
        return response()->json([
            'data'    => $data,
            'message' => $message,
            'status'  => $status,
        ], $status);
    }

    protected function error(string $message = 'Error', array $errors = [], int $status = 400): JsonResponse
    {
        $payload = [
            'message' => $message,
            'status'  => $status,
        ];

        if (!empty($errors)) {
            $payload['errors'] = $errors;
        }

        return response()->json($payload, $status);
    }

    protected function paginated(mixed $resource): JsonResponse
    {
        return response()->json([
            'data'    => $resource->items(),
            'message' => 'Success',
            'status'  => 200,
            'meta'    => [
                'current_page' => $resource->currentPage(),
                'per_page'     => $resource->perPage(),
                'total'        => $resource->total(),
                'last_page'    => $resource->lastPage(),
            ],
        ]);
    }
}
```

- [ ] **Step 4: Create AuthController**

Create `app/Http/Controllers/Api/AuthController.php`:
```php
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\User;
use App\Traits\ApiResponse;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\ValidationException;

class AuthController extends Controller
{
    use ApiResponse;

    public function login(Request $request): JsonResponse
    {
        $request->validate([
            'email'    => ['required', 'email'],
            'password' => ['required', 'string'],
        ]);

        if (!Auth::attempt($request->only('email', 'password'))) {
            return $this->error('Invalid credentials', [], 401);
        }

        $request->session()->regenerate();

        $user = Auth::user();

        return $this->success([
            'id'          => $user->id,
            'name'        => $user->name,
            'email'       => $user->email,
            'roles'       => $user->getRoleNames(),
            'permissions' => $user->getAllPermissions()->pluck('name'),
        ], 'Login successful');
    }

    public function logout(Request $request): JsonResponse
    {
        Auth::guard('web')->logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return $this->success(null, 'Logged out successfully');
    }

    public function me(Request $request): JsonResponse
    {
        $user = $request->user();

        return $this->success([
            'id'          => $user->id,
            'name'        => $user->name,
            'email'       => $user->email,
            'roles'       => $user->getRoleNames(),
            'permissions' => $user->getAllPermissions()->pluck('name'),
        ]);
    }
}
```

Create `Modules/Masters/Http/Controllers/Api/LookupController.php`:
```php
<?php

namespace Modules\Masters\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Traits\ApiResponse;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Modules\Masters\Http\Resources\LookupResource;

class LookupController extends Controller
{
    use ApiResponse;

    private array $modelMap = [
        'banks'           => \Modules\Masters\Models\Bank::class,
        'vendors'         => \Modules\Masters\Models\Vendor::class,
        'regions'         => \Modules\Masters\Models\Region::class,
        'locations'       => \Modules\Masters\Models\Location::class,
        'designations'    => \Modules\Masters\Models\Designation::class,
        'amenity-types'   => \Modules\Masters\Models\AmenityType::class,
        'unit-types'      => \Modules\Masters\Models\UnitType::class,
        'payment-methods' => \Modules\Masters\Models\PaymentMethod::class,
        'countries'       => \Modules\Masters\Models\Country::class,
        'nationalities'   => \Modules\Masters\Models\Nationality::class,
    ];

    private function resolveModel(Request $request): string
    {
        // Determine lookup type from route name segment
        // Route names are e.g. 'banks.index', 'regions.store', etc.
        $routeName = $request->route()->getName();
        $prefix    = Str::before($routeName, '.');

        return $this->modelMap[$prefix] ?? throw new \RuntimeException("Unknown lookup: {$prefix}");
    }

    public function index(Request $request): JsonResponse
    {
        $modelClass = $this->resolveModel($request);
        $perPage    = $request->integer('per_page', 100);
        $search     = $request->string('search');

        $items = $modelClass::query()
            ->when($search, fn($q) => $q->where('name', 'ilike', "%{$search}%"))
            ->latest()
            ->paginate($perPage);

        return $this->paginated(LookupResource::collection($items)->resource);
    }

    public function store(Request $request): JsonResponse
    {
        $modelClass = $this->resolveModel($request);

        $data = $request->validate([
            'name' => ['required', 'string', 'max:255'],
        ]);

        $item = $modelClass::create($data);

        return $this->success(new LookupResource($item), 'Created successfully', 201);
    }

    public function show(Request $request, int $id): JsonResponse
    {
        $modelClass = $this->resolveModel($request);
        $item       = $modelClass::findOrFail($id);

        return $this->success(new LookupResource($item));
    }

    public function update(Request $request, int $id): JsonResponse
    {
        $modelClass = $this->resolveModel($request);
        $item       = $modelClass::findOrFail($id);

        $data = $request->validate([
            'name' => ['sometimes', 'string', 'max:255'],
        ]);

        $item->update($data);

        return $this->success(new LookupResource($item), 'Updated successfully');
    }

    public function destroy(Request $request, int $id): JsonResponse
    {
        $modelClass = $this->resolveModel($request);
        $item       = $modelClass::findOrFail($id);
        $item->delete();

        return $this->success(null, 'Deleted successfully');
    }
}
```

```typescript
import api, { initCsrf } from '@/lib/api'
import { useAuthStore } from '@/store/authStore'
import { useRouter } from 'next/navigation'
import type { LoginCredentials, User } from '@/types/auth'
import type { ApiResponse } from '@/types/api'

export function useAuth() {
  const setUser   = useAuthStore((s) => s.setUser)
  const clearUser = useAuthStore((s) => s.clearUser)
  const router    = useRouter()

  const login = async (credentials: LoginCredentials): Promise<void> => {

    setUser(response.data.data)

  const logout = async (): Promise<void> => {
    await api.post('/api/v1/auth/logout')
    clearUser()
    router.push('/login')
  }

  const fetchMe = async (): Promise<User> => {
    const response = await api.get<ApiResponse<User>>('/api/v1/auth/me')
    return response.data.data
  }

  return { login, logout, fetchMe }
}
```

```typescript
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

const PUBLIC_PATHS = ['/login']

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl

  // Check for auth session via Zustand-persisted cookie key
  const authStorage = request.cookies.get('plms-auth')
  let isAuthenticated = false

  if (authStorage) {
    try {
      const parsed = JSON.parse(authStorage.value)
      isAuthenticated = !!parsed?.state?.user
    } catch {
      isAuthenticated = false
    }
  }

  const isPublicPath = PUBLIC_PATHS.some((p) => pathname.startsWith(p))

  if (!isAuthenticated && !isPublicPath) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  if (isAuthenticated && pathname === '/login') {
    return NextResponse.redirect(new URL('/masters/buildings', request.url))
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico|api).*)'],
}
```
